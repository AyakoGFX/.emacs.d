* Configure Emacs for LaTeX
Install all the latex pkgs
   #+begin_src sh
     sudo pacman -S texlive
   #+end_src

Next we need to tell Emacs how to handle LaTeX.
#+begin_src emacs-lisp
;; Force Org-mode to export with XeLaTeX
(setq org-latex-compiler "xelatex")

;; Set XeLaTeX as the build process (runs twice to resolve TOC and references)
(setq org-latex-pdf-process
      '("xelatex -shell-escape -interaction nonstopmode -output-directory %o %f"
        "xelatex -shell-escape -interaction nonstopmode -output-directory %o %f"))

(setq org-latex-listings 't)

(require 'ox-latex)

;; Custom Class Definition
(add-to-list 'org-latex-classes
             '("org-plain-latex"
               "\\documentclass{article}
               [NO-DEFAULT-PACKAGES]
               [PACKAGES]
               [EXTRA]"
               ("\\section{%s}" . "\\section*{%s}")
               ("\\subsection{%s}" . "\\subsection*{%s}")
               ("\\subsubsection{%s}" . "\\subsubsection*{%s}")
               ("\\paragraph{%s}" . "\\paragraph*{%s}")
               ("\\subparagraph{%s}" . "\\subparagraph*{%s}")))

(put 'upcase-region 'disabled nil)
#+end_src
